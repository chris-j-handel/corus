    return (
        bi_co_surfacing,
        [(bi_offering, co_bi_co_bi_co_corusing, bi_co_bi_co_bi_torusing, bi_co_inseparating)
         for bi_offering, (co_bi_co_bi_co_corusing, bi_co_bi_co_bi_torusing, bi_co_inseparating)
         in co_bi_carrying.items()])
